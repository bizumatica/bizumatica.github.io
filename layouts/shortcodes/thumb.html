{{- /* Shortcode: {{< thumb src="imagem.jpg" pos="left" legenda="Miniatura" alt="Texto Alt" >}} Sintaxe Simplificada (Posicional): {{< thumb "imagem.jpg" "right" "Legenda" >}} Parâmetros: - src / [0]: Caminho da imagem, URL externa ou recurso local (fallback: .Page.Params.image) - pos / [1]: Alinhamento (left | right | center) - Default: "left" - legenda / [2]: Texto de legenda sob a moldura - alt / [3]: Texto alternativo para acessibilidade (fallback: .Page.Title) */ -}} {{- $srcRaw := .Get "src" | default (.Get 0) | default .Page.Params.image -}} {{- $pos := .Get "pos" | default (.Get 1) | default "left" -}} {{- $legenda := .Get "legenda" | default (.Get 2) -}} {{- $alt := .Get "alt" | default (.Get 3) | default .Page.Title -}} {{- /* Resolução de Imagem: Suporte a Page Resources e Assets Globais */ -}} {{- $src := "" -}} {{- if $srcRaw -}} {{- if hasPrefix $srcRaw "http" -}} {{- $src = $srcRaw -}} {{- else -}} {{- $res := .Page.Resources.GetMatch $srcRaw -}} {{- if not $res -}} {{- $res = resources.Get $srcRaw -}} {{- end -}} {{- if $res -}} {{- $src = $res.RelPermalink -}} {{- else -}} {{- $src = $srcRaw -}} {{- end -}} {{- end -}} {{- end -}} {{- if $src -}} {{- /* Trava de Injeção Única de CSS via Page Scratch (Zero Duplicação) */ -}} {{- if not (.Page.Scratch.Get "thumb_css_loaded") -}} {{- .Page.Scratch.Set "thumb_css_loaded" true -}} {{- end -}} {{- end -}}
