{{/* recomenda.html — cartão de recomendação agnóstico (livro, documentação, vídeo, ferramenta, link...) Uso: {{< recomenda id="rocky-livros" >}}Descrição opcional em Markdown.{{< /recomenda >}} {{< recomenda titulo="..." autor="..." tipo="..." link="..." >}}...{{< /recomenda >}} Precedência de cada campo: parâmetro do shortcode > entrada no data/recomendacoes.yaml > padrão. Campos (todos opcionais, exceto que é preciso ter ao menos titulo, via parâmetro ou YAML): id chave no recomendacoes.yaml (mapa) ou valor do campo "id" (lista) titulo título do item autor autor, editora ou organização (omitido se vazio) tipo "Livro Didático", "Documentação Oficial", "Vídeo", "Ferramenta"... (define o ícone) nivel "Iniciante", "Avançado"... (omitido se vazio) idioma "Inglês", "Português"... (omitido se vazio) isbn gera link e capa via Open Library quando link/capa não forem informados link URL do material (tem prioridade sobre o ISBN) capa URL ou caminho local (/img/capa.jpg) da imagem; sem ele, usa a capa do ISBN, se houver cor cor de destaque em hex (#rgb ou #rrggbb); inválida ou ausente = cor derivada do id/título rotulo texto pequeno do topo (padrão: "Recomendação") botao texto do botão (padrão: "Acessar material") icone força o ícone: livro | doc | video | ferramenta | link descricao texto usado quando o shortcode não tem corpo */}} {{/* 1. Resolução de dados (Hugo 0.156+ / hugo.Data) */}} {{- $id := .Get "id" -}} {{- $entry := dict -}} {{- $data := hugo.Data.recomendacoes -}} {{- if and $id $data -}} {{- if reflect.IsMap $data -}} {{- $entry = index $data $id | default dict -}} {{- else if reflect.IsSlice $data -}} {{- with where $data "id" "eq" $id -}} {{- $entry = index . 0 -}} {{- end -}} {{- end -}} {{- end -}} {{/* 2. Cascata de campos: shortcode > YAML > padrão */}} {{- $titulo := .Get "titulo" | default (index $entry "titulo") | default "Recomendação" -}} {{- $autor := .Get "autor" | default (index $entry "autor") | default "" -}} {{- $tipo := .Get "tipo" | default (index $entry "tipo") | default "" -}} {{- $nivel := .Get "nivel" | default (index $entry "nivel") | default "" -}} {{- $idioma := .Get "idioma" | default (index $entry "idioma") | default "" -}} {{- $rotulo := .Get "rotulo" | default (index $entry "rotulo") | default "Recomendação" -}} {{- $botao := .Get "botao" | default (index $entry "botao") | default "Acessar material" -}} {{- $iconeReq := .Get "icone" | default (index $entry "icone") | default "" -}} {{- $descricao := .Get "descricao" | default (index $entry "descricao") | default "" -}} {{/* ISBN pode vir como número no YAML; normaliza para só dígitos (e X) */}} {{- $isbn := .Get "isbn" | default (index $entry "isbn") | default "" | string | replaceRE `[^0-9Xx]` "" -}} {{/* 3. Cor de destaque: valida hex e cai para a paleta determinística */}} {{- $palette := slice "#7aa2f7" "#bb9af7" "#7dcfff" "#9ece6a" "#e0af68" "#f7768e" "#05d9e8" "#ff2a6d" -}} {{- $seed := $id | default $titulo | string -}} {{- $cor := index $palette (mod (hash.FNV32a $seed) (len $palette)) -}} {{- $corReq := .Get "cor" | default (index $entry "cor") | default "" | string -}} {{- if findRE `^#([0-9a-fA-F]{3}|[0-9a-fA-F]{6})$` $corReq -}} {{- $cor = $corReq -}} {{- end -}} {{/* 4. Link: explícito > Open Library via ISBN */}} {{- $link := .Get "link" | default (index $entry "link") | default "" -}} {{- if and (not $link) $isbn -}} {{- $link = printf "https://openlibrary.org/isbn/%s" $isbn -}} {{- end -}} {{- $externo := hasPrefix $link "http" -}} {{/* 5. Capa: explícita > Open Library via ISBN (?default=false faz a API responder 404 sem capa, ativando o fallback) */}} {{- $capa := .Get "capa" | default (index $entry "capa") | default "" -}} {{- if and (not $capa) $isbn -}} {{- $capa = printf "https://covers.openlibrary.org/b/isbn/%s-M.jpg?default=false" $isbn -}} {{- end -}} {{- $capaSrc := "" -}} {{- if $capa -}} {{- if hasPrefix $capa "/" -}} {{- $capaSrc = $capa | relURL -}} {{- else -}} {{- $capaSrc = $capa -}} {{- end -}} {{- end -}} {{/* 6. Ícone: forçado por "icone" ou inferido do "tipo" */}} {{- $icons := dict "livro" `` "doc" `` "video" `` "ferramenta" `` "link" `` -}} {{- $key := lower $iconeReq -}} {{- if not (index $icons $key) -}} {{- $t := lower $tipo -}} {{- if or (strings.Contains $t "livro") (strings.Contains $t "leitura") (strings.Contains $t "didático") (strings.Contains $t "didatico") -}} {{- $key = "livro" -}} {{- else if or (strings.Contains $t "document") (strings.Contains $t "guia") (strings.Contains $t "artigo") (strings.Contains $t "paper") (strings.Contains $t "manual") -}} {{- $key = "doc" -}} {{- else if or (strings.Contains $t "vídeo") (strings.Contains $t "video") (strings.Contains $t "curso") (strings.Contains $t "aula") (strings.Contains $t "podcast") -}} {{- $key = "video" -}} {{- else if or (strings.Contains $t "ferrament") (strings.Contains $t "software") (strings.Contains $t "aplicativo") (strings.Contains $t "utilitár") -}} {{- $key = "ferramenta" -}} {{- else -}} {{- $key = "link" -}} {{- end -}} {{- end -}} {{- $svg := printf `` (index $icons $key) | safeHTML -}} {{/* 7. Corpo: conteúdo do shortcode > campo descricao do YAML */}} {{- $corpo := trim (printf "%s" .Inner) " \n\r\t" -}} {{- if not $corpo -}} {{- $corpo = $descricao -}} {{- end -}} {{/* 8. CSS único por página */}} {{- if not (.Page.Store.Get "recomenda-css-loaded") -}} {{- .Page.Store.Set "recomenda-css-loaded" true -}} {{- end -}} {{/* 9. Markup semântico */}}
